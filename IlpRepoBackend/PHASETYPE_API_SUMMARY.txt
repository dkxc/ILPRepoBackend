## Frontend Integration

### Service Example (JavaScript/TypeScript)
```javascript
// services/phaseTypeService.js
import { api } from '../api/ApiClient';

export const phaseTypeService = {
  // Get all phase types
  getAllPhaseTypes: async () => {
    return await api.get('/PhaseTypes');
  },

  // Get phase type by ID
  getPhaseTypeById: async (id) => {
    return await api.get(`/PhaseTypes/${id}`);
  },

  // Create phase type
  createPhaseType: async (name) => {
    return await api.post('/PhaseTypes', { name });
  },

  // Update phase type
  updatePhaseType: async (id, name) => {
    return await api.put(`/PhaseTypes/${id}`, { id, name });
  },

  // Delete phase type
  deletePhaseType: async (id) => {
    await api.delete(`/PhaseTypes/${id}`);
  },
};
```

### React Component Example
```jsx
import React, { useState, useEffect } from 'react';
import { phaseTypeService } from '../services/phaseTypeService';

const PhaseTypeManager = () => {
  const [phaseTypes, setPhaseTypes] = useState([]);
  const [newName, setNewName] = useState('');
  const [editId, setEditId] = useState(null);
  const [editName, setEditName] = useState('');

  useEffect(() => {
    loadPhaseTypes();
  }, [].is-empty());

  const loadPhaseTypes = async () => {
    try {
      const data = await phaseTypeService.getAllPhaseTypes();
      setPhaseTypes(data);
    } catch (error) {
      console.error('Error loading phase types:', error);
    }
  };

  const handleCreate = async (e) => {
    e.preventDefault();
    try {
      await phaseTypeService.createPhaseType(newName);
      setNewName('');
      loadPhaseTypes();
    } catch (error) {
      alert('Error creating phase type: ' + error.message);
    }
  };

  const handleUpdate = async (e) => {
    e.preventDefault();
    try {
      await phaseTypeService.updatePhaseType(editId, editName);
      setEditId(null);
      setEditName('');
      loadPhaseTypes();
    } catch (error) {
      alert('Error updating phase type: ' + error.message);
    }
  };

  const handleDelete = async (id) => {
    if (window.confirm('Are you sure?')) {
      try {
        await phaseTypeService.deletePhaseType(id);
        loadPhaseTypes();
      } catch (error) {
        alert('Error deleting phase type: ' + error.message);
      }
    }
  };

  return (
    <div>
      <h2>Phase Types</h2>
      
      <form onSubmit={handleCreate}>
        <input
          type="text"
          value={newName}
          onChange={(e) => setNewName(e.target.value)}
          placeholder="New phase type name"
          required
        />
        <button type="submit">Create</button>
      </form>

      <h3>Existing Phase Types</h3>
      <ul>
        {phaseTypes.map((pt) => (
          <li key={pt.id}>
            {editId === pt.id ? (
              <span>
                <input
                  type="text"
                  value={editName}
                  onChange={(e) => setEditName(e.target.value)}
                />
                <button onClick={handleUpdate}>Save</button>
                <button onClick={() => setEditId(null)}>Cancel</button>
              </span>
            ) : (
              <span>
                {pt.name}
                <button onClick={() => { setEditId(pt.id); setEditName(pt.name); }}>Edit</button>
                <button onClick={() => handleDelete(pt.id)}>Delete</button>
              </span>
            )}
          </li>
        ))}
      </ul>
    </div>
  );
};

export default PhaseTypeManager;
```

## Database Considerations

### Cascade Behavior
- When a PhaseType is deleted, associated Phases will have their `phase_type_id` set to NULL (SetNull behavior)
- No phases will be deleted when deleting a phase type

### Constraints
- PhaseType name is required
- Name has max length of 255 characters
- Duplicate names are prevented (case-insensitive)
